GET basico
	# sistema
	snmpget -v2c -c public 127.0.0.1:1161 1.3.6.1.2.1.1.1.0
	

	# manager (DisplayString)
	snmpget -v2c -c public 127.0.0.1:1161 1.3.6.1.4.1.28308.1.1.0
	# cpuUsage (Integer32)
	snmpget -v2c -c public 127.0.0.1 1.3.6.1.4.1.28308.1.3.0
	
GETNEXT
	# Start just before your subtree’s first instance (.1.0)
	snmpgetnext -v2c -c public 127.0.0.1:1161 1.3.6.1.4.1.28308.1.1
	
SET
	# Change manager and email (strings)
	snmpset -v2c -c private 127.0.0.1 1.3.6.1.4.1.28308.1.1.0 s "Alice"
	snmpset -v2c -c private 127.0.0.1 1.3.6.1.4.1.28308.1.2.0 s "alice@example.com"

	# Change threshold (int)
	snmpset -v2c -c private 127.0.0.1 1.3.6.1.4.1.28308.1.4.0 i 75
